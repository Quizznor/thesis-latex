\printbibheading

\defbibnote{auger}{%
Non-public sources from within the Pierre Auger collaboration are listed under 
\sc{A}.}

\defbibnote{software}{%
The list of software and frameworks used throughout this work is indexed with 
\sc{S}.}

\defbibnote{private}{%
Non-public sources from personal correspondence are grouped under the prefix 
\sc{C}.}

\defbibnote{other}{%
All public (non-physics or software related) reference are marked with the label
\sc{O}.}

\defbibnote{physics}{%
Physical references, including from the \PAO, can be found under the label 
\sc{P}.}
